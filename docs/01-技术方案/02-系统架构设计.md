class UDPMotionController:
    """运动主机UDP控制器"""
    
    def __init__(self, ip: str = "192.168.1.103", port: int = 43893):
        self.target_addr = (ip, port)
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    
    def send_heartbeat(self):
        """发送心跳包"""
        cmd = struct.pack('<III', 0x21040001, 0, 0)
        self.sock.sendto(cmd, self.target_addr)
    
    def stand_up(self):
        """起立"""
        self._send_command(0x21010202)
    
    def sit_down(self):
        """趴下"""
        self._send_command(0x21010202)
    
    def emergency_stop(self):
        """软急停"""
        self._send_command(0x21020C0E)
    
    def enter_ai_mode(self):
        """进入AI模式"""
        self._send_command(0x21010528)
    
    def exit_ai_mode(self):
        """退出AI模式"""
        self._send_command(0x2101052B)
    
    def _send_command(self, code: int):
        """发送简单指令"""
        cmd = struct.pack('<III', code, 0, 0)
        self.sock.sendto(cmd, self.target_addr)
```

class Inspector:
    """巡检主控制器"""
    
    def __init__(self, config: Dict):
        self.config = config
        self.crack_detector = CrackDetector()
        self.temp_monitor = TemperatureMonitor()
        self.udp_controller = UDPMotionController()
        self.gateway = WebSocketGateway(config["server_url"])
        self.checkpoint_manager = CheckpointManager(config["waypoints"])
    
    async def start_inspection(self):
        """启动巡检任务"""
        await self.udp_controller.enter_ai_mode()
        
        for waypoint in self.checkpoint_manager.waypoints:
            # 移动到航点
            await self._move_to_waypoint(waypoint)
            
            # 执行检测
            if waypoint.type == "CRACK_INSPECTION":
                await self._inspect_crack(waypoint)
            elif waypoint.type == "TEMPERATURE_MONITOR":
                await self._monitor_temperature(waypoint)
            
            # 等待下一航点
            await asyncio.sleep(waypoint.wait_time)
    
    async def _inspect_crack(self, waypoint: Waypoint):
        """执行裂缝检测"""
        # 广角检测
        crack = await self.crack_detector.detect(wide_angle_image)
        
        if crack and crack["confidence"] > 0.5:
            # 触发变焦
            await self.ptz_controller.zoom_to(10)
            await asyncio.sleep(0.5)  # 等待稳定
            
            # 精细测量
            measurement = await self.crack_detector.measure(
                zoomed_image, crack["bbox"]
            )
            
            # 上报结果
            await self.gateway.send_inspection_result({
                "type": "crack",
                "width_mm": measurement["width_mm"],
                "length_mm": measurement["length_mm"],
                "confidence": measurement["confidence"],
                "waypoint_id": waypoint.id
            })
```

## 三、数据模型

### 3.1 检测结果模型

```python
from dataclasses import dataclass
from typing import Optional
from enum import Enum

class DefectType(Enum):
    CRACK = "crack"
    HONEYCOMB = "honeycomb"

class AlertLevel(Enum):
    NORMAL = "normal"
    WARN = "warn"
    CRITICAL = "critical"

@dataclass
class CrackResult:
    """裂缝检测结果"""
    defect_type: DefectType
    bbox: Tuple[int, int, int, int]  # (x, y, w, h)
    confidence: float
    width_mm: Optional[float] = None
    length_mm: Optional[float] = None
    
@dataclass
class TemperatureAlert:
    """温度告警"""
    alert_level: AlertLevel
    temperature: float
    roi: Tuple[int, int, int, int]
    alert_id: str
    timestamp: datetime
```

### 3.2 SQLite表结构

```sql
-- 检测结果表
CREATE TABLE inspection_results (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    type TEXT NOT NULL,  -- 'crack' | 'temperature'
    timestamp DATETIME NOT NULL,
    waypoint_id TEXT,
    confidence REAL,
    width_mm REAL,
    length_mm REAL,
    temperature REAL,
    alert_level TEXT,
    snapshot_url TEXT,
    uploaded INTEGER DEFAULT 0
);

-- 航点表
CREATE TABLE waypoints (
    id TEXT PRIMARY KEY,
    x REAL NOT NULL,
    y REAL NOT NULL,
    theta REAL NOT NULL,
    type TEXT NOT NULL,  -- 'CRACK_INSPECTION' | 'TEMPERATURE_MONITOR'
    wait_time REAL DEFAULT 0.0
);

-- 告警历史表
CREATE TABLE alert_history (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    alert_id TEXT UNIQUE NOT NULL,
    type TEXT NOT NULL,
    level TEXT NOT NULL,
    timestamp DATETIME NOT NULL,
    details TEXT  -- JSON格式
);
```
