```python
from src.perception.temperature_monitor import TemperatureMonitor

monitor = TemperatureMonitor()

# 模拟热成像数据（512x640，单位℃）

thermal_frame = np.full((512, 640), 40.0, dtype=np.float32)
thermal_frame[200:300, 200:300] = 48.0  # 高温区域

result = monitor.check_temperature(thermal_frame)

assert result['max_temperature'] == 48.0

# 测试ROI监测
result = monitor.check_temperature(thermal_frame, roi=(200, 200, 100, 100))
assert result['status'] == 'CRITICAL'  # ROI内最高温50℃触发告警

print("✅ 温度监测模块测试通过")
```
